library(reshape2)

data <- airquality

data$Month <- factor(data$Month,
                     levels = 5:9,
                     labels = c("May", "June", "July", "August", "September"))

melted_data <- melt(
  data,
  id.vars = c("Month", "Day"),
  measure.vars = c("Ozone", "Solar.R", "Wind", "Temp")
)

print(melted_data)

casted_data <- dcast(
  melted_data,
  Month ~ variable,
  fun.aggregate = mean,
  na.rm = TRUE
)

print(casted_data)
